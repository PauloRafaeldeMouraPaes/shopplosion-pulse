CREATE TABLE IF NOT EXISTS public.document_versions (id uuid PRIMARY KEY DEFAULT gen_random_uuid(), document_id uuid NOT NULL REFERENCES public.documents(id) ON DELETE CASCADE, content_fingerprint text NOT NULL, chunk_count integer NOT NULL DEFAULT 0, captured_at timestamptz NOT NULL DEFAULT now(), UNIQUE(document_id, content_fingerprint)); CREATE INDEX IF NOT EXISTS document_versions_document_captured_idx ON public.document_versions(document_id, captured_at DESC); CREATE TABLE IF NOT EXISTS public.intelligence_signal_events (id uuid PRIMARY KEY DEFAULT gen_random_uuid(), signal_id uuid NOT NULL REFERENCES public.intelligence_signals(id) ON DELETE CASCADE, industry_id uuid NOT NULL REFERENCES public.industries(id) ON DELETE CASCADE, fingerprint text NOT NULL, observed_at timestamptz NOT NULL DEFAULT now(), confidence numeric NOT NULL CHECK (confidence >= 0 AND confidence <= 1), direction text NOT NULL DEFAULT 'new' CHECK (direction IN ('new','up','down','stable','resolved')), change_score numeric NOT NULL DEFAULT 0 CHECK (change_score >= 0 AND change_score <= 1), evidence jsonb NOT NULL DEFAULT '[]'::jsonb); CREATE INDEX IF NOT EXISTS intelligence_signal_events_signal_observed_idx ON public.intelligence_signal_events(signal_id, observed_at DESC); CREATE INDEX IF NOT EXISTS intelligence_signal_events_industry_observed_idx ON public.intelligence_signal_events(industry_id, observed_at DESC); ALTER TABLE public.document_versions ENABLE ROW LEVEL SECURITY; ALTER TABLE public.intelligence_signal_events ENABLE ROW LEVEL SECURITY; CREATE POLICY document_versions_select_own_industry ON public.document_versions FOR SELECT TO authenticated USING (EXISTS (SELECT 1 FROM public.documents d WHERE d.id=document_id AND d.industry_id=public.current_industry_id())); CREATE POLICY signal_events_select_own_industry ON public.intelligence_signal_events FOR SELECT TO authenticated USING (industry_id=public.current_industry_id());
