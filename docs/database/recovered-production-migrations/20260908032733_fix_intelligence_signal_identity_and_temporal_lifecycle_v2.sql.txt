CREATE OR REPLACE FUNCTION public.pulse_normalize_signal_key(p_industry uuid,p_signal_type text,p_title text) RETURNS text LANGUAGE sql IMMUTABLE AS $$ SELECT encode(digest(coalesce(p_industry::text,'')||'|'||lower(regexp_replace(coalesce(p_signal_type,''),'[^a-zA-Z0-9]+',' ','g'))||'|'||lower(regexp_replace(coalesce(p_title,''),'[^a-zA-Z0-9]+',' ','g')),'sha256'),'hex') $$;
UPDATE public.intelligence_signals SET signal_key=public.pulse_normalize_signal_key(industry_id,signal_type,title) WHERE signal_key IS NULL;
DROP INDEX IF EXISTS public.intelligence_signals_fingerprint_idx;
DROP INDEX IF EXISTS public.intelligence_signals_fingerprint_unique_idx;
CREATE UNIQUE INDEX IF NOT EXISTS intelligence_signals_industry_signal_key_uidx ON public.intelligence_signals(industry_id,signal_key) WHERE signal_key IS NOT NULL;
CREATE OR REPLACE FUNCTION public.pulse_resolve_stale_intelligence_signals() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path=public AS $function$
DECLARE v_resolved int:=0; BEGIN UPDATE public.intelligence_signals s SET status='resolved',updated_at=now() WHERE s.status='active' AND s.observed_at < now()-interval '3 days' AND NOT EXISTS (SELECT 1 FROM public.intelligence_signal_events e WHERE e.signal_id=s.id AND e.observed_at >= now()-interval '3 days'); GET DIAGNOSTICS v_resolved=ROW_COUNT; RETURN jsonb_build_object('status','completed','resolved',v_resolved); END; $function$;
CREATE OR REPLACE FUNCTION public.pulse_intelligence_quality_check() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path=public AS $function$
DECLARE r_id uuid; v_checked int:=0; v_grounded int:=0; v_evidence_fail int:=0; v_duplicate_fail int:=0; v_pass boolean;
BEGIN INSERT INTO public.intelligence_quality_runs DEFAULT VALUES RETURNING id INTO r_id; SELECT count(*) INTO v_checked FROM public.intelligence_signals s WHERE s.status='active'; SELECT count(*) INTO v_grounded FROM public.intelligence_signals s WHERE s.status='active' AND jsonb_typeof(s.evidence)='array' AND jsonb_array_length(s.evidence)>0 AND cardinality(s.source_document_ids)>0 AND EXISTS (SELECT 1 FROM public.documents d WHERE d.industry_id=s.industry_id AND d.id=ANY(s.source_document_ids)); v_evidence_fail:=v_checked-v_grounded; SELECT count(*) INTO v_duplicate_fail FROM (SELECT industry_id,signal_key,count(*) n FROM public.intelligence_signals WHERE status='active' AND signal_key IS NOT NULL GROUP BY industry_id,signal_key HAVING count(*)>1) q; v_pass:=(v_evidence_fail=0 AND v_duplicate_fail=0); UPDATE public.intelligence_quality_runs SET finished_at=now(),signals_checked=v_checked,grounded_signals=v_grounded,evidence_failures=v_evidence_fail,duplicate_failures=v_duplicate_fail,pass=v_pass WHERE id=r_id; PERFORM public.pulse_resolve_stale_intelligence_signals(); RETURN jsonb_build_object('status','completed','signals_checked',v_checked,'grounded_signals',v_grounded,'evidence_failures',v_evidence_fail,'duplicate_failures',v_duplicate_fail,'pass',v_pass); END; $function$;
