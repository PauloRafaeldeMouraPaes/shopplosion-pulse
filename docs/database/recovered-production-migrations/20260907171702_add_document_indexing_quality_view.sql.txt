CREATE OR REPLACE VIEW public.document_indexing_quality WITH (security_invoker = true) AS SELECT d.industry_id, COUNT(*)::integer AS documents, COUNT(*) FILTER (WHERE lower(COALESCE(d.mime_type,'')) IN ('application/pdf','text/plain','text/csv','application/json','text/markdown'))::integer AS compatible_documents, COUNT(*) FILTER (WHERE dc.id IS NOT NULL)::integer AS indexed_documents, COUNT(*) FILTER (WHERE dc.id IS NULL AND lower(COALESCE(d.mime_type,'')) IN ('application/pdf','text/plain','text/csv','application/json','text/markdown'))::integer AS pending_compatible_documents, COUNT(dc.id)::integer AS chunks, COUNT(dc.id) FILTER (WHERE dc.search_vector IS NOT NULL)::integer AS searchable_chunks FROM public.documents d LEFT JOIN public.document_chunks dc ON dc.document_id=d.id GROUP BY d.industry_id; GRANT SELECT ON public.document_indexing_quality TO authenticated; REVOKE ALL ON public.document_indexing_quality FROM anon;
