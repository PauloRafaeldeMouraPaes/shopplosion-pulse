CREATE OR REPLACE FUNCTION public.pulse_intelligence_worker_token() RETURNS text LANGUAGE sql SECURITY DEFINER SET search_path TO 'vault','pg_catalog' AS $$ SELECT decrypted_secret FROM vault.decrypted_secrets WHERE name='pulse_intelligence_worker_token' LIMIT 1 $$; REVOKE ALL ON FUNCTION public.pulse_intelligence_worker_token() FROM PUBLIC, anon, authenticated; GRANT EXECUTE ON FUNCTION public.pulse_intelligence_worker_token() TO service_role;
