INSERT INTO public.document_version_chunks(version_id,document_id,chunk_index,content_fingerprint,content) SELECT v.id,c.document_id,c.chunk_index,md5(COALESCE(c.content,'')),COALESCE(c.content,'') FROM public.document_versions v JOIN public.document_chunks c ON c.document_id=v.document_id WHERE v.content_fingerprint=md5(COALESCE((SELECT string_agg(c2.chunk_index::text||':'||COALESCE(c2.content,''),E'\\n' ORDER BY c2.chunk_index) FROM public.document_chunks c2 WHERE c2.document_id=v.document_id),'')) ON CONFLICT(version_id,chunk_index) DO UPDATE SET content_fingerprint=EXCLUDED.content_fingerprint,content=EXCLUDED.content;
