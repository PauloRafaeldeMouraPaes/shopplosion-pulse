DROP VIEW IF EXISTS public.intelligence_recent_changes; CREATE VIEW public.intelligence_recent_changes WITH (security_invoker=true) AS SELECT e.id,e.document_id,e.previous_version_id,e.current_version_id,e.added_chunks,e.removed_chunks,e.modified_chunks,e.changed_chunk_count,e.change_score,e.observed_at,d.industry_id,d.filename FROM public.document_change_events e JOIN public.documents d ON d.id=e.document_id; REVOKE ALL ON public.intelligence_recent_changes FROM anon; GRANT SELECT ON public.intelligence_recent_changes TO authenticated;
