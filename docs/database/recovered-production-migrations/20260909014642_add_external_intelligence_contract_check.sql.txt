create or replace function public.pulse_external_intelligence_contract_check() returns jsonb language plpgsql security definer set search_path = public as $$ declare v_signals integer := 0; v_grounded integer := 0; v_events integer := 0; v_failures integer := 0; r record; begin for r in select s.id,s.industry_id,s.evidence from public.intelligence_signals s where s.status='active' and jsonb_typeof(s.evidence)='array' loop v_signals := v_signals + 1; if exists (select 1 from jsonb_array_elements(r.evidence) e join public.intelligence_source_items i on i.id = nullif(e->>'source_item_id','')::uuid where i.industry_id=r.industry_id and i.status='active' and coalesce(length(e->>'quote'),0)>0 and i.content like '%' || (e->>'quote') || '%' and i.fetched_at >= now() - interval '30 days') then v_grounded := v_grounded + 1; else v_failures := v_failures + 1; end if; if exists (select 1 from public.intelligence_signal_events se where se.signal_id=r.id) then v_events := v_events + 1; end if; end loop; return jsonb_build_object('pass',v_failures=0,'signals_checked',v_signals,'grounded_signals',v_grounded,'signals_with_events',v_events,'evidence_failures',v_failures); end; $$; revoke all on function public.pulse_external_intelligence_contract_check() from public; grant execute on function public.pulse_external_intelligence_contract_check() to service_role;
