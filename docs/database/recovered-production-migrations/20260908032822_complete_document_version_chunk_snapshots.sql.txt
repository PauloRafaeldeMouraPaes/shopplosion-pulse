CREATE OR REPLACE FUNCTION public.pulse_capture_document_version(p_document_id uuid) RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path=public AS $function$
DECLARE v_fp text; v_count int; v_version uuid; v_prev uuid; v_added int[]; v_removed int[]; v_modified int[]; v_changed int; v_score numeric;
BEGIN SELECT md5(COALESCE(string_agg(c.chunk_index::text||':'||COALESCE(c.content,''),E'\\n' ORDER BY c.chunk_index),'')),count(*)::int INTO v_fp,v_count FROM public.document_chunks c WHERE c.document_id=p_document_id; IF v_count=0 THEN RETURN NULL; END IF; SELECT id INTO v_version FROM public.document_versions WHERE document_id=p_document_id AND content_fingerprint=v_fp LIMIT 1; IF v_version IS NOT NULL THEN RETURN v_version; END IF; SELECT id INTO v_prev FROM public.document_versions WHERE document_id=p_document_id ORDER BY captured_at DESC LIMIT 1; INSERT INTO public.document_versions(document_id,content_fingerprint,chunk_count) VALUES(p_document_id,v_fp,v_count) RETURNING id INTO v_version; INSERT INTO public.document_version_chunks(version_id,document_id,chunk_index,content_fingerprint,content) SELECT v_version,c.document_id,c.chunk_index,md5(COALESCE(c.content,'')),COALESCE(c.content,'') FROM public.document_chunks c WHERE c.document_id=p_document_id ON CONFLICT(version_id,chunk_index) DO NOTHING; IF v_prev IS NOT NULL THEN SELECT COALESCE(array_agg(cur.chunk_index ORDER BY cur.chunk_index) FILTER (WHERE old.chunk_index IS NULL),ARRAY[]::int[]),COALESCE(array_agg(old.chunk_index ORDER BY old.chunk_index) FILTER (WHERE cur.chunk_index IS NULL),ARRAY[]::int[]),COALESCE(array_agg(cur.chunk_index ORDER BY cur.chunk_index) FILTER (WHERE old.content_fingerprint IS DISTINCT FROM cur.content_fingerprint),ARRAY[]::int[]) INTO v_added,v_removed,v_modified FROM (SELECT chunk_index,content_fingerprint FROM public.document_version_chunks WHERE version_id=v_version) cur FULL JOIN (SELECT chunk_index,content_fingerprint FROM public.document_version_chunks WHERE version_id=v_prev) old USING(chunk_index); v_changed=coalesce(array_length(v_added,1),0)+coalesce(array_length(v_removed,1),0)+coalesce(array_length(v_modified,1),0); v_score=least(1.0,v_changed::numeric/greatest(v_count,(SELECT chunk_count FROM public.document_versions WHERE id=v_prev),1)); INSERT INTO public.document_change_events(document_id,previous_version_id,current_version_id,added_chunks,removed_chunks,modified_chunks,changed_chunk_count,change_score,observed_at) VALUES(p_document_id,v_prev,v_version,v_added,v_removed,v_modified,v_changed,v_score,now()) ON CONFLICT(document_id,current_version_id) DO NOTHING; END IF; RETURN v_version; END; $function$;
