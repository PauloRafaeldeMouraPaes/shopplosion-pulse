DROP FUNCTION IF EXISTS public.auth_industry_context(); CREATE FUNCTION public.auth_industry_context() RETURNS TABLE(industry_id uuid, industry_name text, industry_slug text, industry_status text, role text) LANGUAGE sql STABLE SECURITY DEFINER SET search_path TO 'public' AS $function$ select i.id, i.name, i.slug, i.status, p.role from public.profiles p join public.industries i on i.id = p.industry_id where p.id = auth.uid() limit 1; $function$; REVOKE EXECUTE ON FUNCTION public.auth_industry_context() FROM anon; GRANT EXECUTE ON FUNCTION public.auth_industry_context() TO authenticated;
