CREATE TABLE IF NOT EXISTS public.intelligence_signals (id uuid PRIMARY KEY DEFAULT gen_random_uuid(), industry_id uuid NOT NULL REFERENCES public.industries(id) ON DELETE CASCADE, created_by uuid REFERENCES auth.users(id) ON DELETE SET NULL, signal_type text NOT NULL CHECK (signal_type IN ('trend','change','risk','opportunity','competitor','pricing','channel','demand','other')), title text NOT NULL, summary text NOT NULL, confidence numeric(5,4) CHECK (confidence >= 0 AND confidence <= 1), evidence jsonb NOT NULL DEFAULT '[]'::jsonb, source_document_ids uuid[] NOT NULL DEFAULT '{}'::uuid[], status text NOT NULL DEFAULT 'active' CHECK (status IN ('active','dismissed','resolved')), observed_at timestamptz NOT NULL DEFAULT now(), created_at timestamptz NOT NULL DEFAULT now(), updated_at timestamptz NOT NULL DEFAULT now()); CREATE INDEX IF NOT EXISTS intelligence_signals_industry_observed_idx ON public.intelligence_signals(industry_id, observed_at DESC); CREATE INDEX IF NOT EXISTS intelligence_signals_type_idx ON public.intelligence_signals(industry_id, signal_type, observed_at DESC); ALTER TABLE public.intelligence_signals ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS intelligence_signals_select ON public.intelligence_signals; CREATE POLICY intelligence_signals_select ON public.intelligence_signals FOR SELECT TO authenticated USING (industry_id = (SELECT industry_id FROM public.profiles WHERE id = (SELECT auth.uid()))); DROP POLICY IF EXISTS intelligence_signals_insert ON public.intelligence_signals; CREATE POLICY intelligence_signals_insert ON public.intelligence_signals FOR INSERT TO authenticated WITH CHECK (industry_id = (SELECT industry_id FROM public.profiles WHERE id = (SELECT auth.uid()))); DROP POLICY IF EXISTS intelligence_signals_update ON public.intelligence_signals; CREATE POLICY intelligence_signals_update ON public.intelligence_signals FOR UPDATE TO authenticated USING (industry_id = (SELECT industry_id FROM public.profiles WHERE id = (SELECT auth.uid()))) WITH CHECK (industry_id = (SELECT industry_id FROM public.profiles WHERE id = (SELECT auth.uid()))); CREATE OR REPLACE VIEW public.intelligence_daily_summary WITH (security_invoker=true) AS SELECT industry_id, date_trunc('day', observed_at) AS day, count(*)::int AS signals, count(*) FILTER (WHERE signal_type='trend')::int AS trends, count(*) FILTER (WHERE signal_type='change')::int AS changes, count(*) FILTER (WHERE signal_type='risk')::int AS risks, count(*) FILTER (WHERE signal_type='opportunity')::int AS opportunities, round(avg(confidence),3) AS avg_confidence FROM public.intelligence_signals GROUP BY industry_id,date_trunc('day',observed_at);
