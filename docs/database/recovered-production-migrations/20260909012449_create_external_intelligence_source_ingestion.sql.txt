create table if not exists public.intelligence_sources (
 id uuid primary key default gen_random_uuid(), industry_id uuid not null references public.industries(id) on delete cascade, name text not null check (char_length(name) between 1 and 180), source_type text not null check (source_type in ('rss','json','html')), url text not null check (char_length(url) between 8 and 2000), active boolean not null default true, poll_interval_minutes integer not null default 180 check (poll_interval_minutes between 15 and 10080), last_fetched_at timestamptz, last_success_at timestamptz, last_error text, created_at timestamptz not null default now(), updated_at timestamptz not null default now(), unique(industry_id,url));
create index if not exists intelligence_sources_industry_active_idx on public.intelligence_sources(industry_id,active);
create table if not exists public.intelligence_source_items (
 id uuid primary key default gen_random_uuid(), source_id uuid not null references public.intelligence_sources(id) on delete cascade, industry_id uuid not null references public.industries(id) on delete cascade, external_id text not null, title text not null, url text, published_at timestamptz, content text not null, content_fingerprint text not null, fetched_at timestamptz not null default now(), status text not null default 'active' check (status in ('active','expired','rejected')), created_at timestamptz not null default now(), unique(source_id,external_id));
create index if not exists intelligence_source_items_industry_published_idx on public.intelligence_source_items(industry_id,published_at desc);
create table if not exists public.intelligence_source_fetch_runs (
 id uuid primary key default gen_random_uuid(), source_id uuid not null references public.intelligence_sources(id) on delete cascade, industry_id uuid not null references public.industries(id) on delete cascade, started_at timestamptz not null default now(), finished_at timestamptz, status text not null check (status in ('running','completed','failed')), items_seen integer not null default 0, items_changed integer not null default 0, error text);
create index if not exists intelligence_source_fetch_runs_source_started_idx on public.intelligence_source_fetch_runs(source_id,started_at desc);
alter table public.intelligence_sources enable row level security;
alter table public.intelligence_source_items enable row level security;
alter table public.intelligence_source_fetch_runs enable row level security;
create policy intelligence_sources_select on public.intelligence_sources for select to authenticated using (industry_id=(select public.current_industry_id()));
create policy intelligence_sources_insert on public.intelligence_sources for insert to authenticated with check (industry_id=(select public.current_industry_id()));
create policy intelligence_sources_update on public.intelligence_sources for update to authenticated using (industry_id=(select public.current_industry_id())) with check (industry_id=(select public.current_industry_id()));
create policy intelligence_sources_delete on public.intelligence_sources for delete to authenticated using (industry_id=(select public.current_industry_id()));
create policy intelligence_source_items_select on public.intelligence_source_items for select to authenticated using (industry_id=(select public.current_industry_id()));
create policy intelligence_source_fetch_runs_select on public.intelligence_source_fetch_runs for select to authenticated using (industry_id=(select public.current_industry_id()));
create or replace view public.intelligence_source_health with (security_invoker=true) as select s.id,s.industry_id,s.name,s.source_type,s.url,s.active,s.poll_interval_minutes,s.last_fetched_at,s.last_success_at,s.last_error,count(i.id) filter (where i.status='active') as active_items,max(i.published_at) as latest_published_at,max(i.fetched_at) as latest_item_fetched_at from public.intelligence_sources s left join public.intelligence_source_items i on i.source_id=s.id where s.industry_id=(select public.current_industry_id()) group by s.id,s.industry_id,s.name,s.source_type,s.url,s.active,s.poll_interval_minutes,s.last_fetched_at,s.last_success_at,s.last_error;
create or replace function public.pulse_external_source_schedule_tick() returns bigint language plpgsql security definer set search_path='public','extensions','vault','pg_catalog' as $$ declare token text; request_id bigint; begin select decrypted_secret into token from vault.decrypted_secrets where name='pulse_intelligence_worker_token' limit 1; if token is null or token='' then raise exception 'pulse_intelligence_worker_token_missing'; end if; select net.http_post(url:='https://ppfuygnpgywfpiqxsfys.supabase.co/functions/v1/pulse-external-source-worker',headers:=jsonb_build_object('Content-Type','application/json','x-pulse-cron',token),body:=jsonb_build_object('source','cron','requested_at',now()),timeout_milliseconds:=60000) into request_id; return request_id; end; $$;
revoke execute on function public.pulse_external_source_schedule_tick() from public,anon,authenticated;
