CREATE OR REPLACE VIEW public.intelligence_signal_timeline WITH (security_invoker=true) AS SELECT s.id AS signal_id, s.industry_id, s.signal_type, s.title, s.summary, s.status, s.confidence, s.fingerprint, min(e.observed_at) AS first_seen, max(e.observed_at) AS last_seen, count(e.id)::int AS occurrence_count, (array_agg(e.direction ORDER BY e.observed_at DESC))[1] AS latest_direction, max(e.change_score) AS max_change_score FROM public.intelligence_signals s LEFT JOIN public.intelligence_signal_events e ON e.signal_id=s.id GROUP BY s.id; CREATE OR REPLACE VIEW public.intelligence_change_digest WITH (security_invoker=true) AS SELECT industry_id, date_trunc('day', observed_at) AS day, count(*) FILTER (WHERE direction='new')::int AS new_signals, count(*) FILTER (WHERE direction='up')::int AS accelerating, count(*) FILTER (WHERE direction='down')::int AS decelerating, count(*) FILTER (WHERE direction='stable')::int AS stable, count(*) FILTER (WHERE direction='resolved')::int AS resolved, round(avg(change_score),3) AS avg_change_score FROM public.intelligence_signal_events GROUP BY industry_id,date_trunc('day',observed_at);
