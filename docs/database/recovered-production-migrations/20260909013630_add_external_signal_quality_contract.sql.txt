create or replace function public.pulse_intelligence_external_quality_check() returns jsonb language plpgsql security definer set search_path = public as $$ declare total_signals integer := 0; grounded integer := 0; failures integer := 0; r record; begin for r in select id, evidence, industry_id from public.intelligence_signals where status='active' and jsonb_typeof(evidence)='array' and jsonb_array_length(evidence)>0 loop total_signals := total_signals + 1; if exists (select 1 from jsonb_array_elements(r.evidence) e join public.intelligence_source_items i on i.id = nullif(e->>'source_item_id','')::uuid where i.industry_id=r.industry_id and coalesce(length(e->>'quote'),0)>0 and i.content like '%' || (e->>'quote') || '%') then grounded := grounded + 1; else failures := failures + 1; end if; end loop; return jsonb_build_object('pass', failures=0,'signals_checked',total_signals,'grounded_signals',grounded,'evidence_failures',failures); end; $$; revoke all on function public.pulse_intelligence_external_quality_check() from public; grant execute on function public.pulse_intelligence_external_quality_check() to service_role;
