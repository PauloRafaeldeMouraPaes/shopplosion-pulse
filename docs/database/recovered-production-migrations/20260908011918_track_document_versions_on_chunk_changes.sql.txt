CREATE OR REPLACE FUNCTION public.capture_document_version() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path=public AS $$ DECLARE v_document_id uuid; v_fp text; v_count integer; BEGIN v_document_id:=COALESCE(NEW.document_id,OLD.document_id); SELECT md5(COALESCE(string_agg(c.chunk_index::text||':'||COALESCE(c.content,''),E'\n' ORDER BY c.chunk_index),'')),count(*)::int INTO v_fp,v_count FROM public.document_chunks c WHERE c.document_id=v_document_id; IF v_count>0 THEN INSERT INTO public.document_versions(document_id,content_fingerprint,chunk_count) VALUES(v_document_id,v_fp,v_count) ON CONFLICT(document_id,content_fingerprint) DO NOTHING; END IF; RETURN COALESCE(NEW,OLD); END; $$; DROP TRIGGER IF EXISTS document_chunks_version_capture ON public.document_chunks; CREATE TRIGGER document_chunks_version_capture AFTER INSERT OR UPDATE OF content,chunk_index OR DELETE ON public.document_chunks FOR EACH ROW EXECUTE FUNCTION public.capture_document_version(); INSERT INTO public.document_versions(document_id,content_fingerprint,chunk_count) SELECT d.id,md5(COALESCE(string_agg(c.chunk_index::text||':'||COALESCE(c.content,''),E'\n' ORDER BY c.chunk_index),'')),count(*)::int FROM public.documents d JOIN public.document_chunks c ON c.document_id=d.id GROUP BY d.id ON CONFLICT(document_id,content_fingerprint) DO NOTHING;
